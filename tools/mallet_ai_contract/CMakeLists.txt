cmake_minimum_required(VERSION 3.20)
project(EvanScoreMalletAiContract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
set(SCENE ${CMAKE_CURRENT_LIST_DIR}/../../src/notationscene/qml/MuseScore/NotationScene)
add_executable(mallet-ai-contract main.cpp ${SCENE}/malletassistant.cpp ${SCENE}/malletassistant.h ${SCENE}/processmemory.cpp ${SCENE}/malletplacement.cpp)
target_include_directories(mallet-ai-contract PRIVATE ${SCENE})
target_link_libraries(mallet-ai-contract PRIVATE Qt6::Core Qt6::Network)
qt_add_resources(mallet-ai-contract mallet_ai_setup PREFIX "/evanscore/ai" BASE ${SCENE} FILES ${SCENE}/mallet-ai-setup.ps1)
